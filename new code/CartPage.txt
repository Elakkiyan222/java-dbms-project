public void addToCart(CartItem item, int productId) {
   if (instance == null) {
    System.out.println("CartPage not created yet");
    return;
}
    boolean found = false;
    for (CartItem ci : instance.cartItems) {
        if (ci.name.equals(item.name)) {
            ci.quantity += item.quantity;
            found = true;
            break;
        }
    }

    if (!found) {
        instance.cartItems.add(item);
    }
 try {
    Connection con = DBConnection.getConnection();

    String checkSql = "SELECT quantity FROM cart WHERE user_id=? AND product_id=?";
    PreparedStatement checkPst = con.prepareStatement(checkSql);
    checkPst.setInt(1, UserSession.currentUserId);
    checkPst.setInt(2, productId);

    ResultSet rs = checkPst.executeQuery();

    if (rs.next()) {
        String updateSql = "UPDATE cart SET quantity=? WHERE user_id=? AND product_id=?";
        PreparedStatement updatePst = con.prepareStatement(updateSql);
        updatePst.setInt(1, rs.getInt("quantity") + item.quantity);
        updatePst.setInt(2, UserSession.currentUserId);
        updatePst.setInt(3, productId);
        updatePst.executeUpdate();
    } else {
    
        String insertSql = "INSERT INTO cart(user_id, product_id, quantity) VALUES(?,?,?)";
        PreparedStatement insertPst = con.prepareStatement(insertSql);
        insertPst.setInt(1, UserSession.currentUserId);
        insertPst.setInt(2, productId);
        insertPst.setInt(3, item.quantity);
        insertPst.executeUpdate();
    }

} catch(Exception e) {
    e.printStackTrace();
}

    instance.updateCart();
}
private void updateQuantityInDB(CartItem item, int quantity) {
    try {
        Connection con = DBConnection.getConnection();
        String sql = "UPDATE cart SET quantity=? WHERE user_id=? AND product_id=?";
        PreparedStatement pst = con.prepareStatement(sql);
        pst.setInt(1, quantity);
        pst.setInt(2, UserSession.currentUserId);
        pst.setInt(3, item.productId);
        pst.executeUpdate();
    } catch (Exception e) {
        e.printStackTrace();
    }
}
private void removeItemFromDB(CartItem item) {
    try {
        Connection con = DBConnection.getConnection();

        PreparedStatement pst = con.prepareStatement(sql);

        pst.executeUpdate();
    } catch (Exception e) {
        e.printStackTrace();
    }
}
public JPanel createCartPanel(CartItem item) {

    JPanel panel = new JPanel();
    panel.setPreferredSize(new Dimension(180, 260));
    panel.setBorder(BorderFactory.createLineBorder(Color.BLACK));
    panel.setLayout(new BorderLayout());
    ImageIcon icon = new ImageIcon(item.imagePath);
    Image img = icon.getImage().getScaledInstance(150, 120, Image.SCALE_SMOOTH);
    JLabel imgLabel = new JLabel(new ImageIcon(img));
    imgLabel.setHorizontalAlignment(JLabel.CENTER);
    JLabel nameLabel = new JLabel("<html><center>" + item.name + "<br>₹" + item.price + "</center></html>");
    nameLabel.setHorizontalAlignment(JLabel.CENTER);
    JPanel qtyPanel = new JPanel(new FlowLayout(FlowLayout.CENTER, 5, 5));
    JSpinner qtySpinner = new JSpinner(
        new SpinnerNumberModel(item.quantity, 0, 10000, 1)
    );
    qtySpinner.setPreferredSize(new Dimension(60, 30));
    JSpinner.DefaultEditor editor = (JSpinner.DefaultEditor) qtySpinner.getEditor();
    editor.getTextField().setColumns(3);
    qtyPanel.add(new JLabel("Qty:"));
    qtyPanel.add(qtySpinner);
    JLabel subtotalLabel = new JLabel("Subtotal: ₹" + item.getSubtotal());
    subtotalLabel.setHorizontalAlignment(JLabel.CENTER);
    qtySpinner.addChangeListener(e -> {
        int newQty = (int) qtySpinner.getValue();
        if (newQty == 0) {
            removeItemFromDB(item);
            cartItems.remove(item);
            updateCart();
            return;
        }

        try {
            Connection con = DBConnection.getConnection();
            PreparedStatement pst = con.prepareStatement(
                "SELECT available_qty FROM stock WHERE product_id=?"
            );
            pst.setInt(1, item.productId);
            ResultSet rs = pst.executeQuery();
            int stock = 0;
            if (rs.next()) stock = rs.getInt("available_qty");
            if (newQty > stock) {
                JOptionPane.showMessageDialog(null,
                        "Only " + stock + " available!");
                qtySpinner.setValue(stock);
                newQty = stock;
            }
            item.quantity = newQty;
            updateQuantityInDB(item, newQty);

        } catch (Exception ex) {
            ex.printStackTrace();
        }
        subtotalLabel.setText("Subtotal: ₹" + item.getSubtotal());

        updateCart();
    });
    JPanel bottom = new JPanel(new GridLayout(2, 1));
    bottom.add(qtyPanel);
    bottom.add(subtotalLabel);
    panel.add(imgLabel, BorderLayout.CENTER);
    panel.add(nameLabel, BorderLayout.NORTH);
    panel.add(bottom, BorderLayout.SOUTH);

    return panel;
}
public void updateCart() {
    cartPanel.removeAll();
    int total = 0;

    for (CartItem item : cartItems) {
        JPanel panel = createCartPanel(item);
        cartPanel.add(panel);
        total += item.getSubtotal();
    }
    int rows = (int) Math.ceil(cartItems.size() / 4.0);
    cartPanel.setPreferredSize(new Dimension(900, rows * 270));
    totalLabel.setText("Total: ₹" + total);
    cartPanel.revalidate();
    cartPanel.repaint();
}
public void loadCartFromDB() {
    cartItems.clear();
    try {
        Connection con = DBConnection.getConnection();

        String sql = "SELECT c.product_id, c.quantity, p.product_name, p.price, p.image " +
                     "FROM cart c " +
                     "JOIN product p ON c.product_id = p.product_id " +
                     "WHERE c.user_id=?";

        PreparedStatement pst = con.prepareStatement(sql);
        pst.setInt(1, UserSession.currentUserId);

        ResultSet rs = pst.executeQuery();

        while (rs.next()) {
           CartItem item = new CartItem(
    rs.getInt("product_id"),
    rs.getString("product_name"),
    rs.getInt("price"),
    rs.getInt("quantity"),
    rs.getString("image")
);
            cartItems.add(item);
        }

    } catch(Exception e) {
        e.printStackTrace();
    }

    updateCart();
}

    private void placeOrderAfterPayment() {
    try {
        Connection con = DBConnection.getConnection();

        int total = 0;
        for (CartItem item : cartItems) {
            total += item.getSubtotal();
        }

        String orderSql = "INSERT INTO orders(user_id, total_amount, status) VALUES (?, ?, 'pending')";
        PreparedStatement orderPst = con.prepareStatement(orderSql, Statement.RETURN_GENERATED_KEYS);
        orderPst.setInt(1, UserSession.currentUserId);
        orderPst.setInt(2, total);
        orderPst.executeUpdate();

        ResultSet rs = orderPst.getGeneratedKeys();
        int orderId = 0;
        if (rs.next()) {
            orderId = rs.getInt(1);
        }

        String itemSql = "INSERT INTO order_items(order_id, product_id, quantity, price) VALUES (?, ?, ?, ?)";
        PreparedStatement itemPst = con.prepareStatement(itemSql);

        for (CartItem item : cartItems) {
            itemPst.setInt(1, orderId);
            itemPst.setInt(2, item.productId);
            itemPst.setInt(3, item.quantity);
            itemPst.setInt(4, item.price);
            itemPst.executeUpdate();
        }

        String stockSql = "UPDATE stock SET available_qty = available_qty - ? WHERE product_id=?";
        PreparedStatement stockPst = con.prepareStatement(stockSql);

        for (CartItem item : cartItems) {
            stockPst.setInt(1, item.quantity);
            stockPst.setInt(2, item.productId);
            stockPst.executeUpdate();
        }

        PreparedStatement clearPst = con.prepareStatement("DELETE FROM cart WHERE user_id=?");
        clearPst.setInt(1, UserSession.currentUserId);
        clearPst.executeUpdate();

        cartItems.clear();
        updateCart();

        JOptionPane.showMessageDialog(this, "✅ Payment Successful & Order Placed!");

        generateBillHTML(orderId, total);

    } catch (Exception ex) {
        ex.printStackTrace();
        JOptionPane.showMessageDialog(this, "Error: " + ex.getMessage());
    }
}
 private ImageIcon generateQR(String data) {
    try {
        BitMatrix matrix = new MultiFormatWriter().encode(
                data, BarcodeFormat.QR_CODE, 300, 300);

        BufferedImage image = MatrixToImageWriter.toBufferedImage(matrix);
        return new ImageIcon(image);

    } catch (Exception e) {
        e.printStackTrace();
        return null;
    }
}
