    private void jButton1ActionPerformed(java.awt.event.ActionEvent evt) {                                         

    String name = jTextField1.getText().trim();
    String email = jTextField4.getText().trim();
    String phone = jTextField2.getText().trim();
    String password = String.valueOf(jPasswordField1.getPassword());
    String confirmPassword = String.valueOf(jPasswordField2.getPassword());
    jTextField1.setBorder(BorderFactory.createLineBorder(Color.GRAY));
    jTextField4.setBorder(BorderFactory.createLineBorder(Color.GRAY));
    jTextField2.setBorder(BorderFactory.createLineBorder(Color.GRAY));
    jPasswordField1.setBorder(BorderFactory.createLineBorder(Color.GRAY));
    jPasswordField2.setBorder(BorderFactory.createLineBorder(Color.GRAY));
    if (name.isEmpty() || phone.isEmpty() || password.isEmpty() || confirmPassword.isEmpty()) {
        JOptionPane.showMessageDialog(this, "Username, Phone and Password are mandatory!");
        return;
    }
    if (!phone.matches("\\d{10}")) {
        jTextField2.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
        JOptionPane.showMessageDialog(this, "Phone must be exactly 10 digits!");
        return;
    }
    if (!email.isEmpty()) {
        if (!email.matches("^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$")) {
            jTextField4.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
            JOptionPane.showMessageDialog(this, "Invalid email format!");
            return;
        }
    }
    if (!password.equals(confirmPassword)) {
        jPasswordField1.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
        jPasswordField2.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
        JOptionPane.showMessageDialog(this, "Passwords do not match!");
        return;
    }

    try {
        Connection con = DBConnection.getConnection();
        PreparedStatement checkUser = con.prepareStatement("SELECT * FROM user WHERE name=?");
        checkUser.setString(1, name);
        if (checkUser.executeQuery().next()) {
            jTextField1.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
            JOptionPane.showMessageDialog(this, "Username already exists!");
            return;
        }
        PreparedStatement checkPhone = con.prepareStatement("SELECT * FROM user WHERE phone=?");
        checkPhone.setString(1, phone);
        if (checkPhone.executeQuery().next()) {
            jTextField2.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
            JOptionPane.showMessageDialog(this, "Phone already exists!");
            return;
        }
        if (!email.isEmpty()) {
            PreparedStatement checkEmail = con.prepareStatement("SELECT * FROM user WHERE email=?");
            checkEmail.setString(1, email);
            if (checkEmail.executeQuery().next()) {
                jTextField4.setBorder(BorderFactory.createMatteBorder(2,2,2,2,Color.RED));
                JOptionPane.showMessageDialog(this, "Email already exists!");
                return;
            }
        }
        String sql = "INSERT INTO user (name, email, phone, password) VALUES (?, ?, ?, ?)";
        PreparedStatement pst = con.prepareStatement(sql);

        pst.setString(1, name);
        pst.setString(2, email.isEmpty() ? null : email);
        pst.setString(3, phone);
        pst.setString(4, password);

        pst.executeUpdate();

        JOptionPane.showMessageDialog(this, "Account Created Successfully!");

        new UserLogin().setVisible(true);
        this.dispose();

    } catch (Exception e) {
        e.printStackTrace();
        JOptionPane.showMessageDialog(this, "Error: " + e.getMessage());
    }
    }
